    private let diskCache: DiskCacheManager
    private let memoryPressureSource: DispatchSourceMemoryPressure
    
    init() {
        // メモリ制限設定

        diskCache = DiskCacheManager()

            eventMask: .warning,
            queue: .global(qos: .utility)
        )
        
        setupMemoryWarningHandler()
    }
    
    func image(for url: String) async -> UIImage? {
        // 1. メモリキャッシュから確認
        if let cachedImage = memoryCache.object(forKey: url as NSString) {
            return cachedImage
        }
        
        // 2. ディスクキャッシュから確認
        if let diskImage = await diskCache.image(for: url) {
            memoryCache.setObject(diskImage, forKey: url as NSString)
            return diskImage
        }
        
        // 3. ネットワークから取得
        return await downloadAndCache(url: url)
    }
    
    private func setupMemoryWarningHandler() {
        memoryPressureSource.setEventHandler { [weak self] in
            Task { [weak self] in
                await self?.handleMemoryPressure()
            }
        }
        memoryPressureSource.resume()
    }
    
    private func handleMemoryPressure() {
        memoryCache.removeAllObjects()
        Task { await diskCache.cleanupOldImages() }
    }

// MetalImageProcessor.swift
import MetalPerformanceShaders

actor MetalImageProcessor {
    private let device: MTLDevice
    private let commandQueue: MTLCommandQueue
    private let library: MTLLibrary
    
    init() throws {
        guard let device = MTLCreateSystemDefaultDevice() else {
            throw ProcessorError.metalNotSupported
        }
        
        self.device = device
        self.commandQueue = device.makeCommandQueue()!
        self.library = device.makeDefaultLibrary()!
    }
    
    func processImage(_ image: UIImage, operations: [ImageOperation]) async -> UIImage? {
        return await withCheckedContinuation { continuation in
            guard let cgImage = image.cgImage else {
                continuation.resume(returning: nil)
                return
            }
            
            // Metal テクスチャに変換
            let textureLoader = MTKTextureLoader(device: device)
            guard let inputTexture = try? textureLoader.newTexture(cgImage: cgImage) else {
                continuation.resume(returning: nil)
                return
            }
            
            // GPU で並列処理
            let outputTexture = applyOperationsOnGPU(
                inputTexture: inputTexture,
                operations: operations
            )
            
            // 結果を UIImage に変換
            let resultImage = convertTextureToImage(outputTexture)
            continuation.resume(returning: resultImage)
        }
    }
    
    private func applyOperationsOnGPU(
        inputTexture: MTLTexture,
        operations: [ImageOperation]
    ) -> MTLTexture {
        var currentTexture = inputTexture
        
        for operation in operations {
            currentTexture = applyOperation(operation, to: currentTexture)
        }
        
        return currentTexture
    }
}

// AsyncImageProcessor.swift
actor AsyncImageProcessor {
    private let operationQueue: OperationQueue
    private let processingCache = LRUCache<String, UIImage>(capacity: 50)
    
    init() {
        operationQueue = OperationQueue()
        operationQueue.maxConcurrentOperationCount = 4 // CPU コア数に基づく
        operationQueue.qualityOfService = .userInitiated
    }
    
    func processImage(
        _ image: UIImage,
        operations: [ImageOperation]
    ) async -> Result<UIImage, ImageProcessingError> {
        
        let cacheKey = generateCacheKey(image: image, operations: operations)
        
        // キャッシュ確認
        if let cachedResult = processingCache[cacheKey] {
            return .success(cachedResult)
        }
        
        return await withCheckedContinuation { continuation in
            let operation = ImageProcessingOperation(
                image: image,
                operations: operations
            ) { result in
                // 結果をキャッシュ
                if case .success(let processedImage) = result {
                    self.processingCache[cacheKey] = processedImage
                }
                continuation.resume(returning: result)
            }
            
            operationQueue.addOperation(operation)
        }
    }
}

// ImageProcessingOperation.swift
class ImageProcessingOperation: Operation {
    private let image: UIImage
    private let operations: [ImageOperation]
    private let completion: (Result<UIImage, ImageProcessingError>) -> Void
    
    override func main() {
        guard !isCancelled else { return }
        
        autoreleasepool {
            let result = performImageProcessing()
            DispatchQueue.main.async {
                self.completion(result)
            }
        }
    }
    
    private func performImageProcessing() -> Result<UIImage, ImageProcessingError> {
        // Core Image を使用した効率的な画像処理
        let context = CIContext(options: [
            .workingColorSpace: CGColorSpace(name: CGColorSpace.sRGB)!,
            .outputColorSpace: CGColorSpace(name: CGColorSpace.sRGB)!
        ])
        
        guard let ciImage = CIImage(image: image) else {
            return .failure(.invalidInput)
        }
        
        var currentImage = ciImage
        
        for operation in operations {
            guard !isCancelled else {
                return .failure(.cancelled)
            }
            
            currentImage = operation.apply(to: currentImage)
        }
        
        guard let cgImage = context.createCGImage(currentImage, from: currentImage.extent) else {
            return .failure(.processingFailed)
        }
        
        let result = UIImage(cgImage: cgImage)
        return .success(result)
    }
}

// OptimizedDataManager.swift
actor OptimizedDataManager {
    private let supabase: SupabaseClient
    private let queryCache: QueryCache
    private let batchProcessor: BatchProcessor
    
    init(supabase: SupabaseClient) {
        self.supabase = supabase
        self.queryCache = QueryCache(maxSize: 100)
        self.batchProcessor = BatchProcessor()
    }
    
    func optimizedPostFetch(
        limit: Int = 20,
        offset: Int = 0,
        userId: String? = nil
    ) async throws -> [Post] {
        
        let cacheKey = "posts_\(limit)_\(offset)_\(userId ?? "all")"
        
        // キャッシュ確認
        if let cachedPosts = queryCache.get(key: cacheKey, maxAge: 60) {
            return cachedPosts
        }
        
        // バッチクエリの実行
        let posts = try await batchProcessor.execute { batch in
            // メインクエリ
            let postsQuery = supabase
                .from("posts")
                .select("*")
                .order("created_at", ascending: false)
                .range(from: offset, to: offset + limit - 1)
            
            if let userId = userId {
                postsQuery.eq("user_id", value: userId)
            }
            
            batch.add(postsQuery)
            
            // 関連データの並行取得
            let userIds = posts.map(\.userId).unique()
            batch.add(
                supabase
                    .from("profiles")
                    .select("*")
                    .in("id", values: userIds)
            )
            
            // いいね状態の取得
            if let currentUserId = AuthManager.shared.currentUser?.id {
                batch.add(
                    supabase
                        .from("likes")
                        .select("post_id")
                        .eq("user_id", value: currentUserId)
                        .in("post_id", values: posts.map(\.id))
                )
            }
        }
        
        // 結果をキャッシュ
        queryCache.set(key: cacheKey, value: posts)
        
        return posts
    }
}

// QueryCache.swift
actor QueryCache {
    private struct CacheEntry<T> {
        let value: T
        let timestamp: Date
    }
    
    private var cache: [String: Any] = [:]
    private let maxSize: Int
    
    init(maxSize: Int) {
        self.maxSize = maxSize
    }
    
    func get<T>(key: String, maxAge: TimeInterval) -> T? {
        guard let entry = cache[key] as? CacheEntry<T> else {
            return nil
        }
        
        if Date().timeIntervalSince(entry.timestamp) > maxAge {
            cache.removeValue(forKey: key)
            return nil
        }
        
        return entry.value
    }
    
    func set<T>(key: String, value: T) {
        // LRU 削除ロジック
        if cache.count >= maxSize {
            let oldestKey = cache.keys.first
            cache.removeValue(forKey: oldestKey!)
        }
        
        cache[key] = CacheEntry(value: value, timestamp: Date())
    }
}

        var networkLatency: Double = 0.0
        var cacheHitRate: Double = 0.0
    }

        displayLink?.add(to: .main, forMode: .default)
        
        setupMemoryMonitoring()
        setupNetworkMonitoring()
    }

        metrics.frameRate = calculateFrameRate()
        metrics.memoryUsage = getMemoryUsage()
        metrics.cpuUsage = getCPUUsage()
        
        // パフォーマンス劣化の検出
        if metrics.frameRate < 30 {
            Logger.performance.warning("Low frame rate detected: \(metrics.frameRate)")
            NotificationCenter.default.post(name: .performanceDegraded, object: nil)
        }
    }
    
    private func calculateFrameRate() -> Double {
        // 実装省略
        return 60.0
    }
    
    private func getMemoryUsage() -> Double {
        var info = mach_task_basic_info()
        var count = mach_msg_type_number_t(MemoryLayout<mach_task_basic_info>.size)/4
        
        let kerr: kern_return_t = withUnsafeMutablePointer(to: &info) {
            $0.withMemoryRebound(to: integer_t.self, capacity: 1) {
                task_info(mach_task_self_,
                         task_flavor_t(MACH_TASK_BASIC_INFO),
                         $0,
                         &count)
            }
        }
        
        if kerr == KERN_SUCCESS {
            return Double(info.resident_size) / 1024 / 1024 // MB
        }
        
        return 0
    }
}

struct PerformanceExperiment {

        case optimizedImageLoading = "optimized_image_loading"

        variant: Variant,
        metric: String,
